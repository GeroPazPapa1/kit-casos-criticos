```typescript
export const fetchJson = async <T>(url: string): Promise<T> => {
  const res = await fetch(url)
  if (!res.ok) throw new Error(`${res.status} ${res.statusText} — ${url}`)
  return res.json() as Promise<T>
}
```
This is a verbatim copy of the Kobo Intervenciones api.ts. Do not modify. The em-dash
in the error string is intentional for debugging 404s in production.

Create `Dashboard/src/hooks/` directory (if not present) and write
`Dashboard/src/hooks/useEntregas.ts`:
```typescript
import { useEffect, useState } from 'react'
import { fetchJson } from '@/lib/api'
import type { Entrega } from '@/types/domain'
import type { Meta } from '@/types/meta'

const BASE = './data/dist'

type State =
  | { status: 'loading' }
  | { status: 'error'; error: string }
  | { status: 'ok'; entregas: Entrega[]; meta: Meta }

export const useEntregas = (): State => {
  const [state, setState] = useState<State>({ status: 'loading' })

  useEffect(() => {
    let cancelled = false
    Promise.all([
      fetchJson<Entrega[]>(`${BASE}/entregas.json`),
      fetchJson<Meta>(`${BASE}/meta.json`),
    ])
      .then(([entregas, meta]) => {
        if (!cancelled) setState({ status: 'ok', entregas, meta })
      })
      .catch((err: unknown) => {
        if (!cancelled) setState({ status: 'error', error: String(err) })
      })
    return () => { cancelled = true }
  }, [])

  return state
}
```

Critical details:
- `const BASE = './data/dist'` — the `./` prefix is required for Vercel sub-path
  compatibility. Do not use `/data/dist` (absolute path breaks if app is deployed to a
  sub-directory).
- `let cancelled = false` / `if (!cancelled)` — the cancellation guard prevents calling
  setState on an unmounted component. Do not remove.
- `Promise.all([...])` — both fetches run in parallel. If either fails, the catch
  handler receives the first error.
- Import `type` for Entrega and Meta (type-only import, no runtime cost).
- `@/lib/api` and `@/types/domain` use the path alias configured in vite.config.ts.
